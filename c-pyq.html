3️⃣ Largest of Three Numbers

Write a program to find the largest out of three numbers using the ternary operator.

✅ Answer / Program 
#include <stdio.h>

int main()
{
    int a, b, c, largest;

    printf("Enter three numbers: ");
    scanf("%d %d %d", &a, &b, &c);

    largest = (a > b && a > c) ? a :
              (b > c ? b : c);

    printf("Largest = %d", largest);

    return 0;
}
🖥️ Output
Enter three numbers: 10 25 15 Largest = 25

6️⃣ Student Grade

Write a program to print the grade of a student based on marks of 5 subjects entered by the user.

✅ Answer / Program 
#include <stdio.h>

int main()
{
    float a, b, c, d, e, average;

    printf("Enter marks of 5 subjects: ");
    scanf("%f %f %f %f %f",
          &a, &b, &c, &d, &e);

    average = (a + b + c + d + e) / 5;

    printf("Average = %.2f\n", average);

    if(average >= 90)
        printf("Grade A");
    else if(average >= 75)
        printf("Grade B");
    else if(average >= 60)
        printf("Grade C");
    else if(average >= 50)
        printf("Grade D");
    else
        printf("Grade F");

    return 0;
}
🖥️ Output
Enter marks of 5 subjects: 80 85 75 90 70 Average = 80.00 Grade B

8️⃣ Menu-Driven Program

Write a menu-driven program to check even/odd, positive/negative, square and square root.

✅ Answer / Program 
#include <stdio.h>
#include <math.h>

int main()
{
    int choice, n;

    printf("1. Even or Odd\n");
    printf("2. Positive or Negative\n");
    printf("3. Square\n");
    printf("4. Square Root\n");

    printf("Enter choice: ");
    scanf("%d", &choice);

    printf("Enter number: ");
    scanf("%d", &n);

    switch(choice)
    {
        case 1:
            if(n % 2 == 0)
                printf("Even");
            else
                printf("Odd");
            break;

        case 2:
            if(n >= 0)
                printf("Positive");
            else
                printf("Negative");
            break;

        case 3:
            printf("Square = %d", n * n);
            break;

        case 4:
            printf("Square Root = %.2f", sqrt(n));
            break;

        default:
            printf("Invalid Choice");
    }

    return 0;
}
🖥️ Output
1. Even or Odd 2. Positive or Negative 3. Square 4. Square Root Enter choice: 3 Enter number: 5 Square = 25

🔟 Armstrong Number Series

Write a program to print Armstrong numbers from m to n.

✅ Answer / Program 
#include <stdio.h>

int main()
{
    int m, n, i, temp, digit, sum;

    printf("Enter m and n: ");
    scanf("%d %d", &m, &n);

    printf("Armstrong numbers: ");

    for(i = m; i <= n; i++)
    {
        temp = i;
        sum = 0;

        while(temp != 0)
        {
            digit = temp % 10;
            sum = sum + digit * digit * digit;
            temp = temp / 10;
        }

        if(sum == i)
            printf("%d ", i);
    }

    return 0;
}
🖥️ Output
Enter m and n: 100 500 Armstrong numbers: 153 370 371 407

1️⃣2️⃣ Matrix Operations

Write a program to perform matrix addition, subtraction, multiplication and transpose using switch-case.

✅ Answer / Program 
#include <stdio.h>

int main()
{
    int a[2][2], b[2][2];
    int i, j, choice;

    printf("Enter first matrix:\n");

    for(i = 0; i < 2; i++)
        for(j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);

    printf("Enter second matrix:\n");

    for(i = 0; i < 2; i++)
        for(j = 0; j < 2; j++)
            scanf("%d", &b[i][j]);

    printf("1. Addition\n");
    printf("2. Subtraction\n");

    printf("Enter choice: ");
    scanf("%d", &choice);

    switch(choice)
    {
        case 1:
            for(i = 0; i < 2; i++)
            {
                for(j = 0; j < 2; j++)
                    printf("%d ", a[i][j] + b[i][j]);

                printf("\n");
            }
            break;

        case 2:
            for(i = 0; i < 2; i++)
            {
                for(j = 0; j < 2; j++)
                    printf("%d ", a[i][j] - b[i][j]);

                printf("\n");
            }
            break;

        default:
            printf("Invalid Choice");
    }

    return 0;
}
🖥️ Output
First Matrix: 1 2 3 4 Second Matrix: 5 6 7 8 Enter choice: 1 Addition: 6 8 10 12

1️⃣5️⃣ Call by Value and Call by Reference

Illustrate call by value and call by reference using the example of swapping two numbers.

✅ Answer / Program 
#include <stdio.h>

void swap(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

int main()
{
    int a = 10, b = 20;

    printf("Before Swap: %d %d\n", a, b);

    swap(&a, &b);

    printf("After Swap: %d %d", a, b);

    return 0;
}
🖥️ Output
Before Swap: 10 20 After Swap: 20 10

1️⃣6️⃣ Recursive Factorial

Write a recursive function for computing the factorial of a number.

✅ Answer / Program 
#include <stdio.h>

int factorial(int n)
{
    if(n == 0)
        return 1;

    return n * factorial(n - 1);
}

int main()
{
    int n;

    printf("Enter number: ");
    scanf("%d", &n);

    printf("Factorial = %d", factorial(n));

    return 0;
}
🖥️ Output
Enter number: 5 Factorial = 120

2️⃣1️⃣ Employee File

Create a file named Employee and store records of 15 employees. Each record contains Name, Employee Code, Contact No. and Department.

✅ Answer / Program 
#include <stdio.h>

struct Employee
{
    char name[30];
    int code;
    char contact[15];
    char department[30];
};

int main()
{
    struct Employee e;
    FILE *fp;
    int i;

    fp = fopen("Employee.txt", "w");

    if(fp == NULL)
    {
        printf("File cannot be opened");
        return 1;
    }

    for(i = 1; i <= 15; i++)
    {
        printf("\nEnter employee %d details:\n", i);

        printf("Name: ");
        scanf("%s", e.name);

        printf("Employee Code: ");
        scanf("%d", &e.code);

        printf("Contact No: ");
        scanf("%s", e.contact);

        printf("Department: ");
        scanf("%s", e.department);

        fprintf(fp, "%s %d %s %s\n",
                e.name,
                e.code,
                e.contact,
                e.department);
    }

    fclose(fp);

    printf("\n15 employee records stored successfully.");

    return 0;
}
🖥️ Output
Enter employee 1 details: Name: Nitin Employee Code: 101 Contact No: 9876543210 Department: CSE ... 15 employee records stored successfully.
