SLIDE 04 • IMPORTANT PROGRAMS

💻 Unit 2 Programs

1. Even or Odd 

#include <stdio.h>

int main()
{
    int n;

    printf("Enter number: ");
    scanf("%d", &n);

    if(n % 2 == 0)
        printf("Even");
    else
        printf("Odd");

    return 0;
}
Example Output: Even

2. Largest of Three Numbers 

#include <stdio.h>

int main()
{
    int a, b, c;

    printf("Enter three numbers: ");
    scanf("%d %d %d", &a, &b, &c);

    if(a > b && a > c)
        printf("Largest = %d", a);

    else if(b > c)
        printf("Largest = %d", b);

    else
        printf("Largest = %d", c);

    return 0;
}

SLIDE 07 • PROGRAM

💻 Pointer Program

Question: Write a C program to demonstrate pointers. 

#include <stdio.h>

int main()
{
    int a = 10;
    int *p;

    p = &a;

    printf("Value of a = %d\n", a);
    printf("Value using pointer = %d\n", *p);

    return 0;
}
Value of a = 10
Value using pointer = 10
